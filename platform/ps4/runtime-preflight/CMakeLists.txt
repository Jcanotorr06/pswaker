cmake_minimum_required(VERSION 3.25)
project(pswaker_runtime_preflight LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DOLPHINPS4)
  message(FATAL_ERROR "Set DOLPHINPS4 to the prepared, pinned source tree")
endif()
include(${DOLPHINPS4}/toolchain/ps4-love-style.cmake)
add_executable(runtime_preflight main.cpp ${PS4_RUNTIME_SOURCES})
target_compile_options(runtime_preflight PRIVATE -ftls-model=initial-exec)
target_link_libraries(runtime_preflight PRIVATE SceSysmodule SceSystemService SceVideoOut SceUserService pthread)
ps4_add_eboot(runtime_preflight)
