project(pswaker_host_renderer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
set(TRACY_ENABLE OFF CACHE BOOL "" FORCE)
FetchContent_Declare(tracy SOURCE_DIR "${PLATFORM_ROOT}/src/fetch/tracy/tracy-0.14.1" SOURCE_SUBDIR headers-only)

add_library(tracy_headers INTERFACE)
target_include_directories(tracy_headers INTERFACE "${tracy_SOURCE_DIR}/public")
add_library(Tracy::TracyClient ALIAS tracy_headers)
add_library(TracyClient ALIAS tracy_headers)
add_library(webgpu_dawn STATIC IMPORTED GLOBAL)
file(MAKE_DIRECTORY "${HOST_ROOT}/dawn/gen/include")
set_target_properties(webgpu_dawn PROPERTIES
  IMPORTED_LOCATION "${HOST_ROOT}/dawn/src/dawn/native/libwebgpu_dawn.a"
  INTERFACE_INCLUDE_DIRECTORIES "${PLATFORM_ROOT}/src/dawn/include;${HOST_ROOT}/dawn/gen/include;${PLATFORM_ROOT}/build/dawn/gen/include"
  INTERFACE_LINK_LIBRARIES "dl;pthread")
add_library(dawn::webgpu_dawn ALIAS webgpu_dawn)
add_library(dawn::dawncpp_headers INTERFACE IMPORTED GLOBAL)
set_target_properties(dawn::dawncpp_headers PROPERTIES
  INTERFACE_INCLUDE_DIRECTORIES "${PLATFORM_ROOT}/src/dawn/include;${HOST_ROOT}/dawn/gen/include")
set(AURORA_DAWN_PROVIDER vendor CACHE STRING "" FORCE)
set(AURORA_SDL3_PROVIDER system CACHE STRING "" FORCE)
set(SDL3_DIR "${HOST_ROOT}/sdl" CACHE PATH "" FORCE)
set(FETCHCONTENT_SOURCE_DIR_ABSEIL-CPP "${PLATFORM_ROOT}/src/abseil" CACHE PATH "" FORCE)
include("${PLATFORM_ROOT}/toolchain/fetch-inputs.cmake")
# Host-only zlib/libpng sources from an explicitly supplied host dependency tree.
# The PS4 build uses its rebuilt SDK libraries instead.
set(FETCHCONTENT_SOURCE_DIR_ZLIB "${HOST_ROOT}/repro/_deps/zlib-src" CACHE PATH "Host zlib source" FORCE)
set(FETCHCONTENT_SOURCE_DIR_PNG "${HOST_ROOT}/repro/_deps/png-src" CACHE PATH "Host PNG source" FORCE)
set(AURORA_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(AURORA_ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
option(PSWAKER_MODEL_COLLECTOR "Build the maintainer-only model/material collector" OFF)
set(AURORA_ENABLE_DVD ${PSWAKER_MODEL_COLLECTOR} CACHE BOOL "" FORCE)
set(AURORA_ENABLE_CARD ${PSWAKER_MODEL_COLLECTOR} CACHE BOOL "" FORCE)
set(AURORA_ENABLE_THP ${PSWAKER_MODEL_COLLECTOR} CACHE BOOL "" FORCE)
set(AURORA_CACHE_USE_ZSTD OFF CACHE BOOL "" FORCE)
set(DAWN_ENABLE_VULKAN ON CACHE BOOL "" FORCE)
set(DAWN_ENABLE_DESKTOP_GL OFF CACHE BOOL "" FORCE)
set(DAWN_ENABLE_OPENGLES OFF CACHE BOOL "" FORCE)
if(PSWAKER_GAME_ROOT)
  set(AURORA_SOURCE_DIR "${PSWAKER_GAME_ROOT}/src/aurora")
elseif(NOT AURORA_SOURCE_DIR)
  set(AURORA_SOURCE_DIR "${PLATFORM_ROOT}/src/aurora")
endif()
if(PSWAKER_MODEL_COLLECTOR)
  if(NOT PSWAKER_GAME_ROOT)
    message(FATAL_ERROR "The collector requires a prepared PSWAKER_GAME_ROOT")
  endif()
  add_library(pswaker_disc STATIC "${PSWAKER_GAME_ROOT}/src/ps5-reference/ps5/nod/nod.cpp")
  target_include_directories(pswaker_disc PUBLIC "${PSWAKER_GAME_ROOT}/src/ps5-reference/ps5/nod/include")
  add_library(nod::nod ALIAS pswaker_disc)
  set(AURORA_NOD_PROVIDER vendor CACHE STRING "" FORCE)
  set(COS_AURORA_PREPATCHED ON CACHE BOOL "" FORCE)
  set(COS_AURORA_COMMIT e281999cdb4fee6127525a9a9e97979480db4bf3 CACHE STRING "" FORCE)
  set(FETCHCONTENT_SOURCE_DIR_AURORA "${AURORA_SOURCE_DIR}" CACHE PATH "" FORCE)
  set(COS_ASSETS_DIR "${PSWAKER_GAME_ROOT}/assets/GZLE01" CACHE PATH "" FORCE)
  set(COS_RECOMPCORE_DIR "${PSWAKER_GAME_ROOT}/src/recompcore" CACHE PATH "" FORCE)
  set(COS_RUNTIME_ASSETS ON CACHE BOOL "" FORCE)
  set(COS_EXE_ENTRY pswaker_model_entry CACHE STRING "" FORCE)
  add_library(pswaker_model_entry STATIC model-collector.cpp)
  add_subdirectory("${PSWAKER_GAME_ROOT}/src/switchwaker/native" native)
  # Reuse the selected source-built zlib, including its generated host headers.
  # The upstream native recipe assumes a system development package named z.
  set_property(TARGET cos_dsp_hle PROPERTY LINK_LIBRARIES "fmt::fmt;zlibstatic")
  set_property(TARGET cos_dsp_hle PROPERTY INTERFACE_LINK_LIBRARIES "$<LINK_ONLY:fmt::fmt>;$<LINK_ONLY:zlibstatic>")
  # The selected PS4 renderer omits upstream 60 Hz view-delta extensions.
  # Compile these optional hooks through their existing PS4-disabled branches.
  file(GLOB _fps_hooks "${PSWAKER_GAME_ROOT}/src/switchwaker/native/src/pc/game_hooks/pc_fps60_*.cpp")
  list(APPEND _fps_hooks "${PSWAKER_GAME_ROOT}/src/switchwaker/native/src/pc/runtime/pc_frame.cpp")
  set_source_files_properties(${_fps_hooks}
    DIRECTORY "${PSWAKER_GAME_ROOT}/src/switchwaker/native" PROPERTIES COMPILE_DEFINITIONS "__ORBIS__")
  target_link_libraries(pswaker_model_entry PRIVATE cos_game_headers aurora::core aurora::gx sqlite3
    absl::flat_hash_map xxHash::xxhash)
  target_sources(pswaker_model_entry PRIVATE model-contract.cpp)
  target_include_directories(pswaker_model_entry PRIVATE "${AURORA_SOURCE_DIR}/lib" "${CMAKE_CURRENT_SOURCE_DIR}/../game")
  target_link_options(switchwaker PRIVATE
    "LINKER:--wrap=_ZN6aurora3gfx13find_pipelineERKNS_2gx14PipelineConfigERKNS0_18RenderTargetLayoutE")
  # Match the PS4 stock-texture adapter; never consult the owner's optional packs.
  get_target_property(_pc_sources cos_pc SOURCES)
  list(FILTER _pc_sources EXCLUDE REGEX "/pc_hd_textures[.]cpp$")
  set_property(TARGET cos_pc PROPERTY SOURCES "${_pc_sources}")
  target_sources(cos_pc PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../game/stock-textures.cpp")
  target_include_directories(aurora_core PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../game")
  target_include_directories(aurora_gx PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../game")
  target_include_directories(cos_pc PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../game")
  return()
endif()

    "${CMAKE_CURRENT_SOURCE_DIR}/../game")

    "LINKER:--wrap=wgpuAdapterRequestDevice")

  add_executable(renderer-check main.cpp)
  target_include_directories(renderer-check PRIVATE "${AURORA_SOURCE_DIR}/lib")
  target_link_libraries(renderer-check PRIVATE aurora::core aurora::gx aurora::main aurora::vi)
endif()
