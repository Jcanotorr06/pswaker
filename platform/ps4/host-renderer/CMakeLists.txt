cmake_minimum_required(VERSION 3.25)
project(pswaker_host_renderer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
set(TRACY_ENABLE OFF CACHE BOOL "" FORCE)
FetchContent_Declare(tracy SOURCE_DIR "${PLATFORM_ROOT}/src/fetch/tracy/tracy-0.14.1" SOURCE_SUBDIR headers-only)
FetchContent_MakeAvailable(tracy)
add_library(tracy_headers INTERFACE)
target_include_directories(tracy_headers INTERFACE "${tracy_SOURCE_DIR}/public")
add_library(Tracy::TracyClient ALIAS tracy_headers)
add_library(TracyClient ALIAS tracy_headers)
add_library(webgpu_dawn STATIC IMPORTED GLOBAL)
file(MAKE_DIRECTORY "${HOST_ROOT}/dawn/gen/include")
set_target_properties(webgpu_dawn PROPERTIES
  IMPORTED_LOCATION "${HOST_ROOT}/dawn/src/dawn/native/libwebgpu_dawn.a"
  INTERFACE_INCLUDE_DIRECTORIES "${PLATFORM_ROOT}/src/dawn/include;${HOST_ROOT}/dawn/gen/include;${PLATFORM_ROOT}/build/dawn/gen/include"
  INTERFACE_LINK_LIBRARIES "dl;pthread")
add_library(dawn::webgpu_dawn ALIAS webgpu_dawn)
add_library(dawn::dawncpp_headers INTERFACE IMPORTED GLOBAL)
set_target_properties(dawn::dawncpp_headers PROPERTIES
  INTERFACE_INCLUDE_DIRECTORIES "${PLATFORM_ROOT}/src/dawn/include;${HOST_ROOT}/dawn/gen/include")
set(AURORA_DAWN_PROVIDER vendor CACHE STRING "" FORCE)
set(AURORA_SDL3_PROVIDER system CACHE STRING "" FORCE)
set(SDL3_DIR "${HOST_ROOT}/sdl" CACHE PATH "" FORCE)
set(FETCHCONTENT_SOURCE_DIR_ABSEIL-CPP "${PLATFORM_ROOT}/src/abseil" CACHE PATH "" FORCE)
include("${PLATFORM_ROOT}/toolchain/fetch-inputs.cmake")
# Host-only zlib/libpng sources from an explicitly supplied host dependency tree.
# The PS4 build uses its rebuilt SDK libraries instead.
set(FETCHCONTENT_SOURCE_DIR_ZLIB "${HOST_ROOT}/repro/_deps/zlib-src" CACHE PATH "Host zlib source" FORCE)
set(FETCHCONTENT_SOURCE_DIR_PNG "${HOST_ROOT}/repro/_deps/png-src" CACHE PATH "Host PNG source" FORCE)
set(AURORA_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(AURORA_ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(AURORA_ENABLE_DVD OFF CACHE BOOL "" FORCE)
set(AURORA_ENABLE_CARD OFF CACHE BOOL "" FORCE)
set(AURORA_ENABLE_THP OFF CACHE BOOL "" FORCE)
set(AURORA_CACHE_USE_ZSTD OFF CACHE BOOL "" FORCE)
set(DAWN_ENABLE_VULKAN ON CACHE BOOL "" FORCE)
set(DAWN_ENABLE_DESKTOP_GL OFF CACHE BOOL "" FORCE)
set(DAWN_ENABLE_OPENGLES OFF CACHE BOOL "" FORCE)
if(NOT AURORA_SOURCE_DIR)
  set(AURORA_SOURCE_DIR "${PLATFORM_ROOT}/src/aurora")
endif()
add_subdirectory("${AURORA_SOURCE_DIR}" aurora)
add_executable(renderer-check main.cpp)
target_include_directories(renderer-check PRIVATE "${AURORA_SOURCE_DIR}/lib")
target_link_libraries(renderer-check PRIVATE aurora::core aurora::gx aurora::main aurora::vi)
